// day03_wifi_manager.ino
#include <WiFi.h>
#include "secrets.h"  // WIFI_SSID, WIFI_PASS — add secrets.h to .gitignore

enum WifiState { WIFI_DISCONNECTED, WIFI_CONNECTING, WIFI_CONNECTED, WIFI_WAITING_RETRY };
WifiState wifiState = WIFI_DISCONNECTED;

unsigned long connectStarted = 0;
unsigned long retryAt = 0;
uint8_t retryAttempt = 0;

const unsigned long CONNECT_TIMEOUT_MS = 15000;
const uint8_t MAX_BACKOFF_EXP = 6;  // caps backoff at 64 s

void startWifiConnect() {
  WiFi.mode(WIFI_STA);
  WiFi.disconnect();          // drop any stale association, keep the radio on
  delay(100);                 // short settle time; acceptable here
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  connectStarted = millis();
  wifiState = WIFI_CONNECTING;
  Serial.printf("Connecting to %s ...\n", WIFI_SSID);
}

void loopWifiManager() {
  switch (wifiState) {
    case WIFI_CONNECTED:
      if (WiFi.status() != WL_CONNECTED) {
        Serial.println(F("WiFi lost - reconnecting"));
        retryAttempt = 0;
        wifiState = WIFI_DISCONNECTED;
      }
      break;

    case WIFI_DISCONNECTED:
      startWifiConnect();
      break;

    case WIFI_CONNECTING:
      if (WiFi.status() == WL_CONNECTED) {
        wifiState = WIFI_CONNECTED;
        retryAttempt = 0;
        Serial.printf("Connected! IP: %s RSSI: %d dBm ch:%d\n",
                      WiFi.localIP().toString().c_str(), WiFi.RSSI(), WiFi.channel());
      } else if (millis() - connectStarted > CONNECT_TIMEOUT_MS) {
        WiFi.disconnect();
        if (retryAttempt < MAX_BACKOFF_EXP) retryAttempt++;
        unsigned long backoff = (1UL << retryAttempt) * 1000UL;
        retryAt = millis() + backoff;
        wifiState = WIFI_WAITING_RETRY;
        Serial.printf("Connect timeout - retry in %lu ms\n", backoff);
      }
      break;

    case WIFI_WAITING_RETRY:
      if ((long)(millis() - retryAt) >= 0) {   // rollover-safe
        wifiState = WIFI_DISCONNECTED;
      }
      break;
  }
}

void setup() {
  Serial.begin(115200);
  delay(200);
  WiFi.persistent(false);        // don't write credentials to flash on every begin()
  WiFi.setAutoReconnect(false);  // the state machine owns reconnects
}

void loop() {
  loopWifiManager();
  delay(250);
}
